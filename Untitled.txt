<script>
const products=[
{name:"Classic Sneakers",price:1499,category:"Fashion",image:"https://images.unsplash.com/photo-1542291026-7eec264c27ff?w=500"},
{name:"Stylish Watch",price:1999,category:"Accessories",image:"https://images.unsplash.com/photo-1523275335684-37898b6baf30?w=500"},
{name:"Premium Headphones",price:2499,category:"Electronics",image:"https://images.unsplash.com/photo-1505740420928-5e560c06d30e?w=500"},
{name:"Casual Backpack",price:999,category:"Accessories",image:"https://images.unsplash.com/photo-1553062407-98eeb64c6a62?w=500"},
{name:"Fashion Sunglasses",price:799,category:"Fashion",image:"https://images.unsplash.com/photo-1511499767150-a48a237f0083?w=500"},
{name:"Modern Shoes",price:1799,category:"Fashion",image:"https://images.unsplash.com/photo-1549298916-b41d501d3772?w=500"}
];
let cart=[];

function render(){
const q=document.getElementById("search").value.toLowerCase();
document.getElementById("products").innerHTML=products
.filter(p=>p.name.toLowerCase().includes(q))
.map((p,i)=>`
<div class="product">
<img src="${p.image}" alt="${p.name}">
<div class="details">
<h3>${p.name}</h3>
<p>${p.category}</p>
<div class="price">₹${p.price.toLocaleString("en-IN")}</div>
<button onclick="add(${products.indexOf(p)})">Add to Cart</button>
</div></div>`).join("");
}
function add(i){cart.push(products[i]);updateCart();}
function updateCart(){
document.getElementById("count").textContent=cart.length;
document.getElementById("total").textContent="Total: ₹"+cart.reduce((s,p)=>s+p.price,0).toLocaleString("en-IN");
document.getElementById("cartItems").innerHTML=cart.length?
cart.map((p,i)=>`<p>${p.name} — ₹${p.price} <button onclick="removeItem(${i})">Remove</button></p>`).join(""):"Your cart is empty.";
}
function removeItem(i){cart.splice(i,1);updateCart();}
function checkout(){
if(!cart.length){alert("Your cart is empty!");return;}
alert("Your cart is ready! Online payment and order processing need to be connected before accepting real orders.");
}
render();
